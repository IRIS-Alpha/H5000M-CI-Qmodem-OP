# =====================================================================
#  OWRT-DEFAULT.txt —— 显式默认出厂配置（局域网 / Wi-Fi / 系统参数）
# =====================================================================
#
#  【作用】
#  本文件集中管理固件出厂时的全部默认参数（后台地址、系统标识、
#  Wi-Fi 名称/密钥/加密/频宽、国家码、时区等），构建时由
#  WRT-CORE 工作流的「Load Default Settings」步骤读取，并经
#  Scripts/Settings.sh 与 Scripts/Handles.sh 注入固件首次启动脚本。
#
#  【如何自定义】
#  1. 直接修改本文件任意「配置项 = 值」并提交即可，无需改动任何脚本或工作流；
#  2. 工作流 inputs（WRT_IP / WRT_PW / WRT_SSID / WRT_WORD 等）显式传入的
#     值优先于本文件；未显式传入时一律使用本文件默认值；
#  3. 取值请遵循各项注释中的可选项与格式要求：非法值可能导致对应功能
#     不可用（例如加密策略写错会导致 Wi-Fi 无法开启）。
#
#  【注意事项】
#  - 文件编码必须为 UTF-8，行尾必须为 LF（CRLF 会导致部分加载环境失败）；
#  - 配置项命名只能包含字母、数字与下划线，值不要加引号；
#  - 以 # 开头的行是注释，不会被加载；删行 = 使用脚本内置兜底默认值。
# =====================================================================


# ============================ 后台管理 ============================
# 后台管理地址（Web 登录 IPv4 地址）
# 格式：合法 IPv4，如 192.168.1.1 / 10.0.0.1
# 该地址同时作为设备 LAN 口 IP 与 DHCP 网段起点
WRT_IP=192.168.10.1

# 后台登录密码提示文本（仅用于 Release 发布说明展示，不影响固件实际密码）
# 固件默认 root 无密码，ImmortalWrt 首次登录时由 LuCI 引导设置；
# 留空（WRT_PW=）即保持无密码
WRT_PW=无


# ============================ 系统标识 ============================
# 系统主机名（LuCI 状态页 / SSH 提示符中显示的名称）
# 格式：仅限字母、数字、下划线、连字符
WRT_NAME=OWRT

# 默认 Web 主题（需已在编译配置中启用对应的 luci-theme-* 与 luci-app-*-config）
# 可选项：aurora（默认）/ argon / material / bootstrap 等
WRT_THEME=aurora


# ============================ Wi-Fi 无线 ============================
# Wi-Fi 默认名称（SSID），2.4G 与 5G 统一使用
# 长度 1~32 字符，建议使用字母与数字，避免中文与特殊符号
WRT_SSID=OWRT

# Wi-Fi 默认密钥（密码）
# 长度至少 8 字符（WPA2 / WPA3 均要求）
WRT_WORD=12345678

# Wi-Fi 加密策略（决定认证方式与终端兼容性）
# 可选项：
#   psk-mixed   WPA / WPA2 混合（默认，兼容新老终端）
#   psk2        WPA2-PSK
#   sae-mixed   WPA3 / WPA2 混合（终端较新时推荐）
#   sae         WPA3-PSK（纯 WPA3，老终端无法连接）
# 说明：构建时脚本会根据驱动栈自动追加必要后缀（如 MTK 闭源栈的
#       +ccmp），此处只需填写裸策略名，无需手动拼接
WIFI_ENCRYPTION=psk-mixed

# 2.4G 频宽（MHz），对应 htmode=HT40
# 可选项：20 / 40（2.4G 使用 HT 模式；40 吞吐更高，20 抗干扰更强）
WIFI_2G_WIDTH=40

# 5G 频宽（MHz），对应 htmode=HE160
# 可选项：20 / 40 / 80 / 160
# 注意：AP3000M（MT7981）硬件上限为 80MHz，配置 160 时构建期会自动降级为 80
WIFI_5G_WIDTH=160


# ============================ 国家与时区 ============================
# 国家 / 地区码（决定可用信道、最大发射功率与合规频段）
# 常见取值：CN（中国，默认）/ US（美国）/ DE（德国）/ GB（英国）/ SG（新加坡）
# 填错可能导致部分信道不可用或发射功率受限
WIFI_COUNTRY=CN

# 系统时区：POSIX 时区串与 IANA 时区名需成对配置
# WRT_TIMEZONE   POSIX 格式，如 CST-8（东八区，固定 8 小时偏移的写法）
# WRT_ZONENAME   IANA 格式，如 Asia/Shanghai（支持夏令时的完整时区名）
# 常用组合：CST-8 + Asia/Shanghai（北京时间）
WRT_TIMEZONE=CST-8
WRT_ZONENAME=Asia/Shanghai
